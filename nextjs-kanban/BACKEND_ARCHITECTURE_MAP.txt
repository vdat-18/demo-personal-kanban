🗄️  DATABASE SCHEMA (5 Models)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  
  Project
  ├─ id, name, description, domain (enum)
  ├─ tasks: Task[] (1:N)
  └─ planningThreads: PlanningThread[] (1:N)

  Task
  ├─ id, title, description
  ├─ status: TaskStatus enum (TODO, IN_PROGRESS, IN_REVIEW, DONE, BLOCKED)
  ├─ priority: TaskPriority enum (LOW, MEDIUM, HIGH, URGENT)
  ├─ projectId: String? (NULLABLE - no cascade)
  ├─ project: Project? (N:1)
  └─ conversations: Conversation[] (1:N)

  Conversation
  ├─ id, taskId, createdAt
  ├─ messages: JSON (not validated)
  ├─ result: JSON (not validated)
  ├─ tokenUsage: JSON (input/output token counts)
  └─ task: Task? (N:1)

  PlanningThread
  ├─ id, projectId, title, createdAt, updatedAt
  ├─ project: Project (N:1, onDelete: Cascade)
  └─ messages: PlanningMessage[] (1:N)

  PlanningMessage
  ├─ id, threadId, role (enum: user|assistant|system)
  ├─ content, tokenUsage: JSON, createdAt
  ├─ thread: PlanningThread (N:1, onDelete: Cascade)
  └─ mutationActions: MutationAction[] (1:N)

  MutationAction
  ├─ id, messageId, mutationIndex, action (enum: approved|dismissed)
  └─ message: PlanningMessage (N:1, onDelete: Cascade)


🔴 CRITICAL CODE SMELLS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  
  SEVERITY: HIGH (Blocking)
  ✗ Schema/validation duplicated inline across route files
    - /api/tasks/route.ts: taskStatusEnum defined
    - /api/tasks/[id]/route.ts: taskStatusEnum redefined
    - /api/conversations/route.ts: schema redefined
    → Should centralize in lib/validations/{domain}.ts

  SEVERITY: MEDIUM (Important)
  ⚠ JSON columns lack type validation
    - conversations.messages: any type
    - conversations.result: any type
    - planning-messages.tokenUsage: any type
    → Add Zod validation schemas + TypeScript interfaces

  ⚠ Task.projectId is nullable but no cascade delete
    - Orphaned tasks possible
    → Add onDelete: Cascade to schema

  ⚠ Agent registry rebuilt per request (/api/agents/stream)
    - Should be cached at process level
    → Singleton pattern with startup initialization

  ⚠ Long route handlers (65+ LOC)
    - Planning message POST mixes DB + agent calls
    → Extract into service layer

  ⚠ Type duplication (TaskStatus defined in 2 places)
    - lib/auto-mode/constants.ts
    - types/domain.ts
    → Single source of truth in types/domain.ts

  ⚠ STATUS_COLORS duplicated with different values
    - constants/status.ts
    - constants/agents.ts
    → Consolidate or clear dependency
